cmake_minimum_required(VERSION 3.16)
project(Bibirble)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Try common locations for wxWidgets on Windows before calling find_package.
# If you have WXWIN set in your environment or wxWidgets is installed to C:/wxWidgets,
# this helps CMake locate it.
if(NOT DEFINED wxWidgets_ROOT_DIR)
    if(DEFINED ENV{WXWIN})
        set(wxWidgets_ROOT_DIR $ENV{WXWIN})
    elseif(EXISTS "C:/wxWidgets")
        set(wxWidgets_ROOT_DIR "C:/wxWidgets")
    endif()
endif()

# Find wxWidgets (explicit COMPONENTS list)
find_package(wxWidgets COMPONENTS core base REQUIRED)
include(${wxWidgets_USE_FILE})

# Find nlohmann_json (for JSON parsing)
find_package(nlohmann_json REQUIRED)

# Source files
set(SOURCES

    src/ChristmasVerses.h

    src/GuessColor.h

# Compile the Bible versions (bible_sections*.json), story_map.json and the app icon into the program (Bibirble.rc
# on Windows, .incbin on Linux) so the game ships as a single file.
if(WIN32)
    list(APPEND SOURCES Bibirble.rc)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # The path goes in as literal text (not a -D macro, which would let
    # predefined macros like `linux` rewrite parts of it).
    set(BIBIRBLE_ICON_FILE "${CMAKE_CURRENT_SOURCE_DIR}/icons/bibirble.png")
    set(BIBIRBLE_STORY_FILE "${CMAKE_CURRENT_SOURCE_DIR}/story_map.json")
    set(BIBIRBLE_WORDS_FILE "${CMAKE_CURRENT_SOURCE_DIR}/data/words.txt")
    # symbol prefix -> file. One verse file per Bible version (src/BibleVersions.h).
    set(BIBIRBLE_BLOBS
        "data|${CMAKE_CURRENT_SOURCE_DIR}/bible_sections.json"
        "kjv|${CMAKE_CURRENT_SOURCE_DIR}/bible_sections_kjv.json"
        "dr|${CMAKE_CURRENT_SOURCE_DIR}/bible_sections_dr.json"
        "cpdv|${CMAKE_CURRENT_SOURCE_DIR}/bible_sections_cpdv.json"
        "icon|${BIBIRBLE_ICON_FILE}"
        "story|${BIBIRBLE_STORY_FILE}"
        "words|${BIBIRBLE_WORDS_FILE}")
    set(BIBIRBLE_BLOB_CPP "${CMAKE_CURRENT_BINARY_DIR}/EmbeddedDataBlob.cpp")
    set(BIBIRBLE_BLOB_ASM "")
    set(BIBIRBLE_BLOB_FILES "")
    foreach(blob IN LISTS BIBIRBLE_BLOBS)
        string(REPLACE "|" ";" parts "${blob}")
        list(GET parts 0 name)
        list(GET parts 1 path)
        list(APPEND BIBIRBLE_BLOB_FILES "${path}")
        string(APPEND BIBIRBLE_BLOB_ASM
            "    \".global bibirble_${name}_start\\n\"\n"
            "    \".global bibirble_${name}_end\\n\"\n"
            "    \"bibirble_${name}_start:\\n\"\n"
            "    \".incbin \\\"${path}\\\"\\n\"\n"
            "    \"bibirble_${name}_end:\\n\"\n")
    endforeach()
    file(WRITE "${BIBIRBLE_BLOB_CPP}"
        "// Generated by CMakeLists.txt: embeds the Bible versions' verse files,\n"
        "// story_map.json, data/words.txt and the app icon\n"
        "// (see src/EmbeddedData.cpp).\n"
        "__asm__(\n"
        "    \".section .rodata\\n\"\n"
        "${BIBIRBLE_BLOB_ASM}"
        "    \".previous\\n\");\n")
    set_source_files_properties("${BIBIRBLE_BLOB_CPP}" PROPERTIES
        OBJECT_DEPENDS "${BIBIRBLE_BLOB_FILES}")
    set_source_files_properties(src/EmbeddedData.cpp PROPERTIES
        COMPILE_DEFINITIONS "BIBIRBLE_HAS_INCBIN=1")
    list(APPEND SOURCES "${BIBIRBLE_BLOB_CPP}")
endif()

# Create executable
add_executable(Bibirble ${SOURCES})

# Link libraries
target_link_libraries(Bibirble ${wxWidgets_LIBRARIES} nlohmann_json::nlohmann_json)

# Story Mode logic tests (no wx needed): cmake --build build --target story_tests && ctest --test-dir build
enable_testing()
add_executable(story_tests
    tests/story_tests.cpp
    src/BibleData.cpp
    src/EmbeddedData.cpp
    src/StoryMap.cpp
    src/StoryProgress.cpp
    src/WordPuzzle.cpp
)
if(DEFINED BIBIRBLE_BLOB_CPP)
    target_sources(story_tests PRIVATE "${BIBIRBLE_BLOB_CPP}")  # EmbeddedData.cpp reads its symbols
endif()
target_include_directories(story_tests PRIVATE src)
target_compile_definitions(story_tests PRIVATE BIBIRBLE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(story_tests nlohmann_json::nlohmann_json)
add_test(NAME story_tests COMMAND story_tests)

# Set startup project on Windows
if(MSVC)
    set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT Bibirble)
endif()
